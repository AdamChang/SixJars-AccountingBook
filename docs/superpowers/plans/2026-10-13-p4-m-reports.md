# P4 段 M：報表 實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL：逐 Task 執行時使用 superpowers:test-driven-development；分派子代理時使用 superpowers:subagent-driven-development；inline 執行時使用 superpowers:executing-plans（ledger 放在 `.superpowers/sdd/2026-10-13-p4-m-reports/progress.md`）。每個 Task 都是「失敗測試 → 確認失敗原因 → 最小實作 → 全綠 → 一個 commit」。

**Goal**：完成 P4 段 M。使用者可以看某個歸屬月份的**月報表**（收入、四種支出性質、財務規劃，依主分類 → 子分類展開，固定／貸款／特別支出另列未付預定，最上方對照月可用餘額，附兩張甜甜圈圖），以及某個曆年的**年報表**（12 個月並列，附年總計、佔收入比、月平均，附每月支出依性質的堆疊長條圖）。同時定案 L 段留下的 Q-A。

**Architecture**：Domain 新增 `Reports/`。歸組規則（spec §6.1、ADR 0008）只寫一次，放在 internal 的 `ReportTree`：輸入是 SQL 端 `GROUP BY` 出來的 `ReportFlow`（歸屬月份 × 交易類型 × 分類 × 對方帳戶 × 財務規劃帳戶 的金額加總），輸出是「每列 N 格」的樹；月報（N = 1）與年報（N = 12）各自是薄薄一層對應（`MonthlyReport.Build`、`YearlyReport.Build`）。`ILedgerSummaryQuery` 新增兩條彙總：`ReportFlowsAsync` 與 `UnpaidPlannedByCategoryAsync`。Application 每個報表一個 query handler，Api 新增 `ReportsEndpoints`。前端新增 `features/reports`（兩個 lazy 路由），Chart.js 只在這兩頁載入，由自寫的 standalone `ChartComponent` 包裝。

**Tech Stack**：.NET 10、EF Core 10（Npgsql）、MediatR、FluentValidation、xUnit v3、FluentAssertions；Angular 22.2、Angular Material 22.2、Chart.js 4.5.1（新增）、Vitest、Playwright。

## Task 1: M1 Domain 的 ReportTree 與 MonthlyReport

**Files:**
- Create: `src/SixJars.Domain/Reports/ReportFlow.cs`、`src/SixJars.Domain/Reports/ReportTree.cs`、`src/SixJars.Domain/Reports/MonthlyReport.cs`
- Create: `tests/SixJars.Domain.Tests/Reports/MonthlyReportTests.cs`

這是整個 M 段的核心：§6.1 的每一條規則都在這裡，而且只寫這一次（D1）。最容易錯的地方：
(1) 支出換號：支出交易是負數，報表要正數，**退款（正數）要抵銷**，不是取絕對值；
(2) 貸款繳款與出資金帶的分類**不參與歸組**（Q-A），寫 `switch` 時很容易順手把有 `CategoryId` 的都丟進分類表；
(3) 「（未分子類）」只在有子分類列時才出現（D6）；
(4) 未付預定要依**主分類**的性質歸到區塊（`Book.ChangeExpenseNature` 只改主分類）。

```csharp
// src/SixJars.Domain/Reports/ReportTree.cs
using SixJars.Domain.Books;
using SixJars.Domain.Common;
using SixJars.Domain.Transactions;

namespace SixJars.Domain.Reports;

public enum ReportSectionKind { Income, Floating, Fixed, Loan, Special }

/// <summary><c>Unassigned</c> 是「（未分子類）」：直接掛在主分類、而該主分類另有子分類列時的金額（spec §6.1）。</summary>
public enum ReportLineKind { Category, Unassigned, LoanAccount, PlanningFund }

/// <summary>樹上的一列；每格是一個月份（月報 1 格、年報 12 格），已換成報表的正數口徑。</summary>
internal sealed record ReportNode(
    ReportLineKind Kind, Guid Id, string Name, decimal[] Values, decimal[]? Principal, IReadOnlyList<ReportNode> Children)
{
    public bool IsEmpty =>
        Children.Count == 0 && Values.All(v => v == 0m) && (Principal is null || Principal.All(p => p == 0m));
}

internal sealed record ReportNodeSection(ReportSectionKind Kind, decimal[] Totals, IReadOnlyList<ReportNode> Lines);

internal sealed record ReportFundNode(PlanningFund Fund, decimal[] Allocation, decimal[] Withdrawal, decimal[] Return);

/// <summary>
/// spec §6.1 的歸組規則（ADR 0008），月報與年報共用（P4 M plan D1）。
/// 貸款繳款的利息分類、出資金的分類<b>不參與</b>歸組（Q-A）：預算 actual 因此恆等於浮動主分類的小計。
/// </summary>
internal sealed class ReportTree
{
    public const string UnassignedName = "（未分子類）";

    private static readonly (ReportSectionKind Section, ExpenseNature Nature)[] ExpenseSections =
    [
        (ReportSectionKind.Floating, ExpenseNature.Floating),
        (ReportSectionKind.Fixed, ExpenseNature.Fixed),
        (ReportSectionKind.Loan, ExpenseNature.Loan),
        (ReportSectionKind.Special, ExpenseNature.Special),
    ];

    private ReportTree(ReportNodeSection income, IReadOnlyList<ReportNodeSection> expenses, IReadOnlyList<ReportFundNode> funds)
    {
        Income = income;
        Expenses = expenses;
        Funds = funds;
    }

    public ReportNodeSection Income { get; }
    public IReadOnlyList<ReportNodeSection> Expenses { get; }
    public IReadOnlyList<ReportFundNode> Funds { get; }

    /// <param name="slotOf">歸屬月份 → 第幾格；不在報表範圍內回傳 null。</param>
    public static ReportTree Build(Book book, IEnumerable<ReportFlow> flows, int width, Func<BudgetMonth, int?> slotOf)
    {
        var byCategory = new Dictionary<CategoryId, decimal[]>();
        var loanTotals = new Dictionary<AccountId, decimal[]>();
        var loanPrincipals = new Dictionary<AccountId, decimal[]>();
        var allocations = new Dictionary<PlanningFundId, decimal[]>();
        var withdrawals = new Dictionary<PlanningFundId, decimal[]>();
        var returns = new Dictionary<PlanningFundId, decimal[]>();

        foreach (var flow in flows)
        {
            if (slotOf(flow.Month) is not { } slot)
            {
                continue;
            }

            switch (flow.Kind)
            {
                case TransactionKind.Income when flow.CategoryId is { } income:
                    Cells(byCategory, income, width)[slot] += flow.Amount;
                    break;
                case TransactionKind.Expense when flow.CategoryId is { } expense:
                    Cells(byCategory, expense, width)[slot] -= flow.Amount;
                    break;
                // 利息分類不參與歸組（Q-A）：一律依貸款帳戶列在「貸款」區
                case TransactionKind.LoanPayment when flow.CounterAccountId is { } loan:
                    Cells(loanTotals, loan, width)[slot] += flow.Amount;
                    Cells(loanPrincipals, loan, width)[slot] += flow.LoanPrincipal;
                    break;
                case TransactionKind.FundAllocation when flow.PlanningFundId is { } fund:
                    Cells(allocations, fund, width)[slot] += flow.Amount;
                    break;
                // 出資金帶的分類不參與歸組（Q-A）
                case TransactionKind.FundWithdrawal when flow.PlanningFundId is { } fund:
                    Cells(withdrawals, fund, width)[slot] += flow.Amount;
                    break;
                case TransactionKind.FundReturn when flow.PlanningFundId is { } fund:
                    Cells(returns, fund, width)[slot] += flow.Amount;
                    break;
            }
        }

        var incomeMains = book.Categories.Where(c => c.IsMain && c.Kind == CategoryKind.Income);
        var income = Section(ReportSectionKind.Income, width, CategoryLines(book, incomeMains, byCategory, width));

        var expenses = new List<ReportNodeSection>();
        foreach (var (section, nature) in ExpenseSections)
        {
            var lines = new List<ReportNode>();
            if (nature == ExpenseNature.Loan)
            {
                // 先列貸款帳戶，再列掛在貸款性質分類上的一般支出（D7）
                lines.AddRange(book.Accounts
                    .Where(a => loanTotals.ContainsKey(a.Id))
                    .OrderBy(a => a.SortOrder)
                    .Select(a => new ReportNode(ReportLineKind.LoanAccount, a.Id.Value, a.Name, loanTotals[a.Id], loanPrincipals[a.Id], []))
                    .Where(n => !n.IsEmpty));
            }

            var mains = book.Categories.Where(c => c.IsMain && c.Kind == CategoryKind.Expense && c.Nature == nature);
            lines.AddRange(CategoryLines(book, mains, byCategory, width));
            expenses.Add(Section(section, width, lines));
        }

        var funds = book.PlanningFunds
            .OrderBy(f => f.SortOrder)
            .Select(f => new ReportFundNode(
                f,
                allocations.GetValueOrDefault(f.Id) ?? new decimal[width],
                withdrawals.GetValueOrDefault(f.Id) ?? new decimal[width],
                returns.GetValueOrDefault(f.Id) ?? new decimal[width]))
            .Where(f => f.Allocation.Concat(f.Withdrawal).Concat(f.Return).Any(v => v != 0m))
            .ToList();

        return new ReportTree(income, expenses, funds);
    }

    /// <summary>支出分類（含子分類）歸到哪個區塊：依<b>主分類</b>的性質；收入分類回傳 null。</summary>
    public static ReportSectionKind? SectionOf(Book book, CategoryId categoryId)
    {
        var category = book.GetCategory(categoryId);
        var main = category.ParentId is { } parentId ? book.GetCategory(parentId) : category;
        return main.Nature switch
        {
            ExpenseNature.Floating => ReportSectionKind.Floating,
            ExpenseNature.Fixed => ReportSectionKind.Fixed,
            ExpenseNature.Loan => ReportSectionKind.Loan,
            ExpenseNature.Special => ReportSectionKind.Special,
            _ => null,
        };
    }

    private static List<ReportNode> CategoryLines(
        Book book, IEnumerable<Category> mains, Dictionary<CategoryId, decimal[]> cells, int width)
    {
        var lines = new List<ReportNode>();
        foreach (var main in mains.OrderBy(c => c.SortOrder))
        {
            var children = book.Categories
                .Where(c => c.ParentId == main.Id)
                .OrderBy(c => c.SortOrder)
                .Select(sub => new ReportNode(
                    ReportLineKind.Category, sub.Id.Value, sub.Name, cells.GetValueOrDefault(sub.Id) ?? new decimal[width], null, []))
                .Where(n => !n.IsEmpty)
                .ToList();
            var direct = cells.GetValueOrDefault(main.Id) ?? new decimal[width];
            // 「（未分子類）」只在有子分類列時才輸出；只有一層時它只是重複主分類的數字（D6）
            if (children.Count > 0 && direct.Any(v => v != 0m))
            {
                children.Add(new ReportNode(ReportLineKind.Unassigned, main.Id.Value, UnassignedName, direct, null, []));
            }

            var values = children.Count > 0 ? Sum(width, children.Select(c => c.Values)) : direct;
            var node = new ReportNode(ReportLineKind.Category, main.Id.Value, main.Name, values, null, children);
            if (!node.IsEmpty)
            {
                lines.Add(node);
            }
        }

        return lines;
    }

    private static ReportNodeSection Section(ReportSectionKind kind, int width, IReadOnlyList<ReportNode> lines) =>
        new(kind, Sum(width, lines.Select(l => l.Values)), lines);

    private static decimal[] Sum(int width, IEnumerable<decimal[]> rows)
    {
        var total = new decimal[width];
        foreach (var row in rows)
        {
            for (var i = 0; i < width; i++)
            {
                total[i] += row[i];
            }
        }

        return total;
    }

    private static decimal[] Cells<TKey>(Dictionary<TKey, decimal[]> map, TKey key, int width) where TKey : notnull
    {
        if (!map.TryGetValue(key, out var cells))
        {
            cells = new decimal[width];
            map[key] = cells;
        }

        return cells;
    }
}
```

```csharp
// src/SixJars.Domain/Reports/MonthlyReport.cs
using SixJars.Domain.Books;
using SixJars.Domain.Common;

namespace SixJars.Domain.Reports;

/// <param name="Amount">正數口徑：收入 = 收入金額、支出 = −支出金額（退款抵銷）、貸款帳戶 = 繳款總額。</param>
/// <param name="Principal">只有貸款帳戶列有值。</param>
/// <param name="Interest">只有貸款帳戶列有值：總額 − 本金。</param>
public sealed record MonthlyReportLine(
    ReportLineKind Kind, Guid Id, string Name, decimal Amount, decimal? Principal, decimal? Interest, IReadOnlyList<MonthlyReportLine> Children);

/// <param name="Unpaid">未付預定支出的小計（不計入 <paramref name="Total"/>）；固定、貸款、特別三區有值（Q17'），其他為 null。</param>
public sealed record MonthlyReportSection(ReportSectionKind Kind, decimal Total, decimal? Unpaid, IReadOnlyList<MonthlyReportLine> Lines);

public sealed record MonthlyPlanningLine(Guid FundId, string Name, decimal Allocation, decimal Withdrawal, decimal Return);

/// <param name="Expenses">固定依浮動、固定、貸款、特別排列（D7）。</param>
/// <param name="ExpenseTotal">四個支出區塊的合計；不含未付預定與財務規劃（ADR 0008）。</param>
public sealed record MonthlyReportResult(
    BudgetMonth Month,
    MonthlyReportSection Income,
    IReadOnlyList<MonthlyReportSection> Expenses,
    decimal ExpenseTotal,
    decimal UnpaidTotal,
    IReadOnlyList<MonthlyPlanningLine> Planning);

/// <summary>某個歸屬月份的月報表（spec §6.1、§6.2）。</summary>
public static class MonthlyReport
{
    /// <param name="unpaidByCategory">該月未付預定支出依分類的預估金額加總（負數），即 <c>ILedgerSummaryQuery.UnpaidPlannedByCategoryAsync</c> 的結果。</param>
    public static MonthlyReportResult Build(
        Book book, BudgetMonth month, IEnumerable<ReportFlow> flows, IReadOnlyDictionary<CategoryId, decimal> unpaidByCategory)
    {
        var tree = ReportTree.Build(book, flows, width: 1, m => m == month ? 0 : null);
        var unpaid = unpaidByCategory
            .Select(p => (Section: ReportTree.SectionOf(book, p.Key), p.Value))
            .Where(p => p.Section is not null)
            .GroupBy(p => p.Section!.Value)
            .ToDictionary(g => g.Key, g => -g.Sum(p => p.Value));

        var expenses = tree.Expenses
            .Select(s => new MonthlyReportSection(
                s.Kind,
                s.Totals[0],
                s.Kind == ReportSectionKind.Floating ? null : unpaid.GetValueOrDefault(s.Kind),
                [.. s.Lines.Select(Line)]))
            .ToList();
        var planning = tree.Funds
            .Select(f => new MonthlyPlanningLine(f.Fund.Id.Value, f.Fund.Name, f.Allocation[0], f.Withdrawal[0], f.Return[0]))
            .ToList();

        return new MonthlyReportResult(
            month,
            new MonthlyReportSection(ReportSectionKind.Income, tree.Income.Totals[0], null, [.. tree.Income.Lines.Select(Line)]),
            expenses,
            expenses.Sum(s => s.Total),
            expenses.Sum(s => s.Unpaid ?? 0m),
            planning);
    }

    private static MonthlyReportLine Line(ReportNode node)
    {
        var amount = node.Values[0];
        var principal = node.Principal?[0];
        return new MonthlyReportLine(
            node.Kind, node.Id, node.Name, amount, principal, principal is { } p ? amount - p : null, [.. node.Children.Select(Line)]);
    }
}
```

```csharp
using SixJars.Domain.Books;
using SixJars.Domain.Common;

namespace SixJars.Domain.Reports;

/// <param name="Months">1–12 月，正數口徑同月報。</param>
/// <param name="ShareOfIncome">年總計 ÷ 收入年總計（小數 4 位）；收入為 0 時為 null（D9）。</param>
/// <param name="MonthlyAverage">年總計 ÷ <see cref="YearlyReportResult.AverageDivisor"/>（小數 2 位）；未來的年份為 null。</param>
public sealed record YearlyFigures(IReadOnlyList<decimal> Months, decimal Total, decimal? ShareOfIncome, decimal? MonthlyAverage);

public sealed record YearlyReportLine(ReportLineKind Kind, Guid Id, string Name, YearlyFigures Figures, IReadOnlyList<YearlyReportLine> Children);

public sealed record YearlyReportSection(ReportSectionKind Kind, YearlyFigures Figures, IReadOnlyList<YearlyReportLine> Lines);

/// <summary>財務規劃依方向分三組（D10），每組依財務規劃帳戶列出。</summary>
public sealed record YearlyPlanning(
    IReadOnlyList<YearlyReportLine> Allocation, IReadOnlyList<YearlyReportLine> Withdrawal, IReadOnlyList<YearlyReportLine> Return);

/// <param name="AverageDivisor">月平均的分母（spec Q10）：過去的年 12、今年 = 本月月份、未來的年 null。</param>
public sealed record YearlyReportResult(
    int Year,
    int? AverageDivisor,
    YearlyReportSection Income,
    IReadOnlyList<YearlyReportSection> Expenses,
    YearlyFigures ExpenseTotal,
    YearlyPlanning Planning);

/// <summary>某個曆年（依歸屬月份的年）的年報表（spec §6.2）：只列實際交易，不列未付預定。</summary>
public static class YearlyReport
{
    /// <param name="today">台北時間的今天所在的月份。</param>
    public static YearlyReportResult Build(Book book, int year, IEnumerable<ReportFlow> flows, BudgetMonth today)
    {
        var tree = ReportTree.Build(book, flows, width: 12, m => m.Year == year ? m.Month - 1 : null);
        int? divisor = year < today.Year ? 12 : year == today.Year ? today.Month : null;
        var incomeTotal = tree.Income.Totals.Sum();

        YearlyFigures Figures(decimal[] months)
        {
            var total = months.Sum();
            return new YearlyFigures(
                months,
                total,
                incomeTotal == 0m ? null : Math.Round(total / incomeTotal, 4, MidpointRounding.AwayFromZero),
                divisor is { } d ? Math.Round(total / d, 2, MidpointRounding.AwayFromZero) : null);
        }

        YearlyReportLine Line(ReportNode node) =>
            new(node.Kind, node.Id, node.Name, Figures(node.Values), [.. node.Children.Select(Line)]);

        YearlyReportSection Section(ReportNodeSection section) =>
            new(section.Kind, Figures(section.Totals), [.. section.Lines.Select(Line)]);

        IReadOnlyList<YearlyReportLine> Funds(Func<ReportFundNode, decimal[]> pick) =>
        [
            .. tree.Funds
                .Where(f => pick(f).Any(v => v != 0m))
                .Select(f => new YearlyReportLine(ReportLineKind.PlanningFund, f.Fund.Id.Value, f.Fund.Name, Figures(pick(f)), [])),
        ];

        var expenseMonths = Enumerable.Range(0, 12).Select(i => tree.Expenses.Sum(s => s.Totals[i])).ToArray();
        return new YearlyReportResult(
            year,
            divisor,
            Section(tree.Income),
            [.. tree.Expenses.Select(Section)],
            Figures(expenseMonths),
            new YearlyPlanning(Funds(f => f.Allocation), Funds(f => f.Withdrawal), Funds(f => f.Return)));
    }
}
```

```csharp
    {
        public void Check(string label, decimal actual, decimal excel)
        {

            if (Math.Round(actual, 2) != Math.Round(expected, 2))
            {
                mismatches.Add($"{prefix}{label}：本系統 {actual:N2}，Excel {expected:N2}，差 {actual - expected:N2}");

        public void Named(string group, IEnumerable<LegacyNamedAmount> excel, IReadOnlyDictionary<string, decimal> system, decimal sign)

            {
                mismatches.Add($"{prefix}{group}：{name} 不在 Excel，本系統 {amount:N2}");
            }
        }
```

```ts
// web/src/app/features/reports/report-rules.ts
import { ChartConfiguration } from 'chart.js';
import { MonthlyReportDto, ReportLineKind, ReportSectionKind, YearlyReportDto } from '../../core/api/dto';

export type ExpenseNatureKind = Exclude<ReportSectionKind, 'Income'>;

export const SECTION_LABELS: Record<ReportSectionKind, string> = {
  Income: '收入', Floating: '浮動支出', Fixed: '固定支出', Loan: '貸款支出', Special: '特別支出',
};
const MONTH_LABELS = Array.from({ length: 12 }, (_, i) => `${i + 1}月`);

interface TreeLine<T> { kind: ReportLineKind; id: string; children: T[] }

export interface VisibleLine<T> { key: string; depth: number; line: T; expandable: boolean; expanded: boolean }

// 主分類與它的「（未分子類）」列 id 相同，鍵要帶 kind
export function lineKey(line: { kind: ReportLineKind; id: string }): string {
  return `${line.kind}:${line.id}`;
}

export function visibleLines<T extends TreeLine<T>>(lines: T[], expanded: ReadonlySet<string>, depth = 0): VisibleLine<T>[] {
  return lines.flatMap(line => {
    const key = lineKey(line);
    const isOpen = expanded.has(key);
    const self: VisibleLine<T> = { key, depth, line, expandable: line.children.length > 0, expanded: isOpen };
    return isOpen ? [self, ...visibleLines(line.children, expanded, depth + 1)] : [self];
  });
}

export function toggle(expanded: ReadonlySet<string>, key: string): Set<string> {
  const next = new Set(expanded);
  if (!next.delete(key)) {
    next.add(key);
  }
  return next;
}

export interface ChartPalette { nature: Record<ExpenseNatureKind, string>; series: string[] }

// 顏色定義在 styles.scss 的 :root（--chart-*），這裡只讀出來交給 Chart.js
export function readChartPalette(style: Pick<CSSStyleDeclaration, 'getPropertyValue'>): ChartPalette {
  const token = (name: string) => style.getPropertyValue(name).trim();
  const series: string[] = [];
  for (let i = 1; token(`--chart-series-${i}`) !== ''; i++) {
    series.push(token(`--chart-series-${i}`));
  }
  return {
    nature: { Floating: token('--chart-floating'), Fixed: token('--chart-fixed'), Loan: token('--chart-loan'), Special: token('--chart-special') },
    series,
  };
}

// 甜甜圈畫不了負數（淨退款），0 也沒有意義：只取正數
export function natureDoughnut(report: MonthlyReportDto, palette: ChartPalette): ChartConfiguration {
  const sections = report.expenses.filter(s => s.total > 0);
  return doughnut(
    sections.map(s => SECTION_LABELS[s.kind]),
    sections.map(s => s.total),
    sections.map(s => palette.nature[s.kind as ExpenseNatureKind]),
  );
}

export function floatingDoughnut(report: MonthlyReportDto, palette: ChartPalette): ChartConfiguration {
  const lines = (report.expenses.find(s => s.kind === 'Floating')?.lines ?? []).filter(l => l.amount > 0);
  return doughnut(
    lines.map(l => l.name),
    lines.map(l => l.amount),
    lines.map((_, i) => palette.series[i % palette.series.length]),
  );
}

export function yearlyStackedBar(report: YearlyReportDto, palette: ChartPalette): ChartConfiguration {
  return {
    type: 'bar',
    data: {
      labels: MONTH_LABELS,
      datasets: report.expenses.map(s => ({
        label: SECTION_LABELS[s.kind],
        data: [...s.figures.months],
        backgroundColor: palette.nature[s.kind as ExpenseNatureKind],
      })),
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      scales: { x: { stacked: true }, y: { stacked: true, beginAtZero: true } },
      plugins: { legend: { position: 'bottom' } },
    },
  };
}

export function formatShare(share: number | null): string {
  return share === null ? '—' : `${(share * 100).toFixed(1)}%`;
}

function doughnut(labels: string[], data: number[], colors: string[]): ChartConfiguration {
  return {
    type: 'doughnut',
    data: { labels, datasets: [{ data, backgroundColor: colors }] },
    options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'bottom' } } },
  };
}
```
